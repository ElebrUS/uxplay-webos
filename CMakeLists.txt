cmake_minimum_required(VERSION 3.16)
project(uxplay_webos LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL REQUIRED sdl2)
set(DEPS "${CMAKE_SOURCE_DIR}/.cache/deps" CACHE PATH "Private ARM dependencies")
option(WEBOS14 "Build the webOS 1.4 software backend" ON)
option(WEBOS14_LGNC "Build the experimental LGNC video backend" OFF)
option(WEBOS14_STARFISH "Build the experimental Rockhopper Starfish backend" OFF)
option(BUILD_RECEIVER "Build the AirPlay receiver (requires tools/build-deps.sh)" ON)
if(WEBOS14)
    add_library(lg_media STATIC src/media_sdl.cpp src/shell.cpp)
    target_compile_definitions(lg_media PUBLIC WEBOS14)
    target_include_directories(lg_media PRIVATE "${DEPS}/include")
    target_link_libraries(lg_media PUBLIC "${DEPS}/lib/libavcodec.a" "${DEPS}/lib/libavutil.a" m dl)
    if(WEBOS14_LGNC)
        target_sources(lg_media PRIVATE src/media_lgnc.cpp)
        target_compile_definitions(lg_media PRIVATE WEBOS14_LGNC)
        target_include_directories(lg_media PRIVATE "${CMAKE_SYSROOT}/usr/include/lgncopenapi")
    endif()
    if(WEBOS14_STARFISH)
        pkg_check_modules(STARFISH_DEPS REQUIRED glib-2.0 luna-service2)
        target_sources(lg_media PRIVATE src/media_starfish.cpp)
        target_compile_definitions(lg_media PRIVATE WEBOS14_STARFISH)
        target_include_directories(lg_media PRIVATE "${CMAKE_SYSROOT}/usr/include/starfish-media-pipeline"
                                                   ${STARFISH_DEPS_INCLUDE_DIRS})
        target_link_directories(lg_media PRIVATE ${STARFISH_DEPS_LIBRARY_DIRS})
        target_link_libraries(lg_media PUBLIC ${STARFISH_DEPS_LIBRARIES})
    endif()
else()
    pkg_check_modules(TTF REQUIRED SDL2_ttf)
    pkg_check_modules(SMP REQUIRED libplayerAPIs)
    pkg_check_modules(ACB REQUIRED AcbAPI)
    add_library(lg_media STATIC src/media.cpp src/shell.cpp)
    target_include_directories(lg_media PRIVATE ${TTF_INCLUDE_DIRS} ${SMP_INCLUDE_DIRS} ${ACB_INCLUDE_DIRS})
    target_link_directories(lg_media PUBLIC ${TTF_LIBRARY_DIRS} ${SMP_LIBRARY_DIRS} ${ACB_LIBRARY_DIRS})
    target_link_libraries(lg_media PUBLIC ${TTF_LIBRARIES} ${SMP_LIBRARIES} ${ACB_LIBRARIES})
    add_executable(media-probe src/probe.cpp)
    target_link_libraries(media-probe PRIVATE lg_media)
    set_target_properties(media-probe PROPERTIES INSTALL_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH TRUE)
endif()
target_compile_definitions(lg_media PUBLIC _GLIBCXX_USE_CXX11_ABI=0)
target_include_directories(lg_media PUBLIC src ${SDL_INCLUDE_DIRS})
target_link_directories(lg_media PUBLIC ${SDL_LIBRARY_DIRS})
target_link_libraries(lg_media PUBLIC ${SDL_LIBRARIES} pthread rt)
target_compile_options(lg_media PRIVATE -Wall -Wextra)
if(BUILD_RECEIVER)
    set(UX "${CMAKE_SOURCE_DIR}/vendor/uxplay")
    add_subdirectory(${UX}/lib/llhttp)
    add_subdirectory(${UX}/lib/playfair)
    add_subdirectory(${UX}/lib/mdnsd)
    file(GLOB CORE CONFIGURE_DEPENDS "${UX}/lib/*.c")
    add_library(airplay STATIC ${CORE})
    target_compile_definitions(airplay PRIVATE PLIST_210 PLIST_230 OPENSSL_API_COMPAT=0x10101000L _REENTRANT _LARGEFILE64_SOURCE _FILE_OFFSET_BITS=64)
    target_include_directories(airplay PUBLIC "${UX}/lib" PRIVATE "${UX}/lib/llhttp" "${UX}/lib/playfair" "${DEPS}/include")
    target_link_libraries(airplay PUBLIC playfair dnssd llhttp "${DEPS}/lib/libplist-2.0.a" "${DEPS}/lib/libcrypto.a" pthread dl m)
    if(EXISTS "${DEPS}/lib/libjitterentropy.a")
        target_link_libraries(airplay PUBLIC "${DEPS}/lib/libjitterentropy.a")
    endif()
    add_executable(uxplay-webos src/main.cpp src/receiver.cpp src/audio.cpp src/h264.cpp)
    option(BUILD_DIRECT_EXPERIMENT "Build the isolated Safari MP4 experiment" OFF)
    if(WEBOS14 AND BUILD_DIRECT_EXPERIMENT)
        target_compile_definitions(uxplay-webos PRIVATE DIRECT_EXPERIMENT)
        target_sources(uxplay-webos PRIVATE src/direct_clip.cpp src/hls_playlist.cpp)
        target_link_libraries(uxplay-webos PRIVATE "${DEPS}/lib/libavformat.a" "${DEPS}/lib/libssl.a" "${DEPS}/lib/libcrypto.a")
    endif()
    target_include_directories(uxplay-webos PRIVATE "${DEPS}/include")
    target_link_libraries(uxplay-webos PRIVATE lg_media airplay "${DEPS}/lib/libavcodec.a" "${DEPS}/lib/libswresample.a" "${DEPS}/lib/libavutil.a")
    # LG's player loads its own older OpenSSL/FFmpeg indirectly. Keep private
    # static symbols out of the dynamic namespace to avoid interposition.
    target_link_options(uxplay-webos PRIVATE -Wl,--exclude-libs,ALL)
    set_target_properties(uxplay-webos PROPERTIES INSTALL_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH TRUE)
endif()

option(BUILD_LGNC_PROBE "Build the standalone LGNC H.264 experiment" OFF)
if(BUILD_LGNC_PROBE)
    add_executable(lgnc-probe tests/lgnc_probe.c)
    target_include_directories(lgnc-probe PRIVATE "${CMAKE_SYSROOT}/usr/include/lgncopenapi")
    # Rockhopper's LGNC library imports dlopen without declaring libdl in DT_NEEDED.
    target_link_libraries(lgnc-probe PRIVATE lgncopenapi -Wl,--no-as-needed dl pthread rt)
endif()
